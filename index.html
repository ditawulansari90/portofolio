function doPost(e) { try { var data = JSON.parse(e.postData.contents); // ID Folder Google Drive Ibu var folderId = "1zULZtPK-AfSxjvzMOtiyfAp2DQRWOFLV"; var folder = DriveApp.getFolderById(folderId); // Simpan Foto Lembar Jawaban ke Google Drive var contentType = data.fileType || "image/jpeg"; var decodedFile = Utilities.base64Decode(data.fileBase64); var blob = Utilities.newBlob(decodedFile, contentType, data.namaSiswa + " - " + data.kelas + " - " + data.namaTugas + ".jpg"); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var fileUrl = file.getUrl(); // Catat ke Google Sheets var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); sheet.appendRow([ new Date(), data.namaSiswa, data.kelas, data.namaTugas, fileUrl // Link foto di Google Drive ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
